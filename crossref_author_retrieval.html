Author retrieval and overlap detection

Retrieving author data from CrossRef

This code retrieves author information from CrossRef for original and replication studies, then checks for author overlap between them.

Note: This script should be run periodically (similar to crossref_ref_retrieval.qmd) to update the author data for new entries. The results are stored in crossref_retrieved_authors.xlsx and crossref_author_overlap.xlsx, which are then loaded by the COS report to add the author_overlap variable to the dataset.

    dois_batch <- missing_dois[i:min(i+99, length(missing_dois))]
    names(dois_batch) <- rep("doi", length(dois_batch))
    
    message(sprintf("  Batch %d: %d DOIs", ceiling(i/100), length(dois_batch)))
    
    res <- cr_works(filter = dois_batch, limit = 100)
    
    if (!is.null(res$data) && nrow(res$data) > 0) {
      batch_authors <- res$data %>%
        select(doi, author) %>%
        mutate(doi = tolower(doi)) %>%
        filter(!is.na(author))
      
      authors_new <- bind_rows(authors_new, batch_authors)
    }
  }
  
  # Convert author lists to JSON strings for storage
  authors_new <- authors_new %>%
    rowwise() %>%
    mutate(authors = as.character(jsonlite::toJSON(author, auto_unbox = TRUE))) %>%
    ungroup() %>%
    select(doi, authors) %>%
    mutate(across(everything(), as.character))

  # Combine with existing data
  authors_df <- authors_df %>%
    mutate(across(everything(), as.character)) %>%
    bind_rows(authors_new) %>%
    distinct(doi, .keep_all = TRUE)
  
  message(sprintf("Retrieved author data for %d new DOIs", nrow(authors_new)))
} else {
  message("No new DOIs to retrieve")

Compute author overlap

# Function to extract author names from JSON
# Returns a vector of lowercased family names for comparison
# More robust handling of various CrossRef author formats
extract_author_names <- function(author_json) {
  # Extract family names from CrossRef author JSON data
  # CrossRef consistently returns dataframes with "family", "given", "sequence" columns
  if (is.na(author_json) || author_json == "" || is.null(author_json)) {
    return(character(0))
  }

  tryCatch({
    authors <- jsonlite::fromJSON(author_json)

    # Handle empty author lists (e.g., from OSF preprints)
    if (is.null(authors) || nrow(authors) == 0) {
      return(character(0))
    }

    # Extract family names (normalized: lowercase, trimmed)
    if ("family" %in% names(authors)) {
      names_extracted <- authors$family %>%
        as.character() %>%
        trimws() %>%
        tolower() %>%
        unique() %>%
        .[!is.na(.) & nzchar(.)]

      return(names_extracted)
    }

    return(character(0))

  }, error = function(e) {
    return(character(0))
  })
}

# Compute author overlap for each study pair
# Also track extraction quality for debugging
author_overlap <- study_pairs %>%
  left_join(authors_df %>% rename(doi_o = doi, authors_o = authors), by = "doi_o") %>%
  left_join(authors_df %>% rename(doi_r = doi, authors_r = authors), by = "doi_r") %>%
  rowwise() %>%
  mutate(
    authors_o_names = list(extract_author_names(authors_o)),
    authors_r_names = list(extract_author_names(authors_r)),
    # Flag extraction status for quality tracking
    o_extracted = length(authors_o_names) > 0,
    r_extracted = length(authors_r_names) > 0,
    both_extracted = o_extracted & r_extracted,
    author_overlap = length(intersect(authors_o_names, authors_r_names)) > 0
  ) %>%
  ungroup()

# Generate quality report
extraction_quality <- author_overlap %>%
  summarise(
    total_pairs = n(),
    both_have_data = sum(!is.na(authors_o) & !is.na(authors_r)),
    both_extracted = sum(both_extracted, na.rm = TRUE),
    extraction_rate = if (sum(!is.na(authors_o) & !is.na(authors_r)) > 0) {
      100 * sum(both_extracted, na.rm = TRUE) / sum(!is.na(authors_o) & !is.na(authors_r))
    } else {
      0
    }
  )
