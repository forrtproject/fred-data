Author retrieval and overlap detection

Retrieving author data from CrossRef

This code retrieves author information from CrossRef for original and replication studies, then checks for author overlap between them.

Note: This script should be run periodically (similar to crossref_ref_retrieval.qmd) to update the author data for new entries. The results are stored in crossref_retrieved_authors.xlsx and crossref_author_overlap.xlsx, which are then loaded by the COS report to add the author_overlap variable to the dataset.

# Function to normalize names to ASCII (remove diacritics)
normalize_to_ascii <- function(text) {
  # Convert to NFD (decomposed form), remove combining marks, convert back to ASCII
  normalized <- iconv(text, from = "UTF-8", to = "ASCII//TRANSLIT")
  # Handle cases where transliteration results in special characters
  normalized <- gsub("[^a-zA-Z0-9\\s]", "", normalized)
  return(normalized)
}

# Function to extract author identifiers from JSON
# Returns a dataframe with family name, first initial (and full given name for debugging)
# Handles character encoding inconsistencies and requires both family name and first initial for matching
extract_author_identifiers <- function(author_json) {
  # Extract author identifiers from CrossRef author JSON data
  # CrossRef consistently returns dataframes with "family", "given", "sequence" columns
  if (is.na(author_json) || author_json == "" || is.null(author_json)) {
    return(data.frame(family = character(), first_initial = character(), stringsAsFactors = FALSE))
  }

  tryCatch({
    authors <- jsonlite::fromJSON(author_json)

    # Handle empty author lists (e.g., from OSF preprints)
    if (is.null(authors) || nrow(authors) == 0) {
      return(data.frame(family = character(), first_initial = character(), stringsAsFactors = FALSE))
    }

    # Extract family names and given names
    if ("family" %in% names(authors)) {
      author_identifiers <- data.frame(
        family = authors$family %>% as.character() %>% trimws(),
        given = if ("given" %in% names(authors)) {
          authors$given %>% as.character() %>% trimws()
        } else {
          rep("", length(authors$family))
        },
        stringsAsFactors = FALSE
      )

      # Normalize family names to ASCII and lowercase
      author_identifiers <- author_identifiers %>%
        mutate(
          family_normalized = tolower(normalize_to_ascii(family)),
          # Extract first initial from given name (before space or first char)
          first_initial = ifelse(
            nzchar(given),
            tolower(substr(given, 1, 1)),
            ""
          ),
          # Create combined identifier: family_given_initial
          author_id = paste0(family_normalized, "_", first_initial)
        ) %>%
        filter(nzchar(family_normalized) & nzchar(first_initial)) %>%
        distinct(author_id, .keep_all = TRUE) %>%
        select(family_normalized, first_initial, author_id)

      return(author_identifiers)
    }

    return(data.frame(family = character(), first_initial = character(), stringsAsFactors = FALSE))

  }, error = function(e) {
    return(data.frame(family = character(), first_initial = character(), stringsAsFactors = FALSE))
  })
}

# Now requires matching on both family name AND first initial
author_overlap <- study_pairs %>%
  left_join(authors_df %>% rename(doi_o = doi, authors_o = authors), by = "doi_o") %>%
  left_join(authors_df %>% rename(doi_r = doi, authors_r = authors), by = "doi_r") %>%
  rowwise() %>%
  mutate(
    authors_o_ids = list(extract_author_identifiers(authors_o)),
    authors_r_ids = list(extract_author_identifiers(authors_r)),
    # Flag extraction status for quality tracking
    o_extracted = nrow(authors_o_ids) > 0,
    r_extracted = nrow(authors_r_ids) > 0,
    both_extracted = o_extracted & r_extracted,
    # Check for overlap: must match on both family name AND first initial
    author_overlap = if (both_extracted) {
      length(intersect(authors_o_ids$author_id, authors_r_ids$author_id)) > 0
    } else {
      NA
    }
  ) %>%
  ungroup()

# Generate quality report
extraction_quality <- author_overlap %>%
  summarise(
    total_pairs = n(),
    both_have_data = sum(!is.na(authors_o) & !is.na(authors_r)),
    both_extracted = sum(both_extracted, na.rm = TRUE),
    extraction_rate = if (sum(!is.na(authors_o) & !is.na(authors_r)) > 0) {
      100 * sum(both_extracted, na.rm = TRUE) / sum(!is.na(authors_o) & !is.na(authors_r))
    } else {
      0
    }
  )
