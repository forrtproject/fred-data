    doi_o = gsub("^doi:", "", doi_o),

  ) %>% # remove redundant DOI urls
  mutate(
    url_r = if_else(!is.na(doi_r) & str_detect(tolower(url_r), if_else(is.na(doi_r), fixed("dummy_val"), fixed(doi_r))), 
                    NA_character_, url_r)
  )

# Normalize URLs before dedup so that e.g. https://osf.io/abc/ and osf.io/abc match
norm_url <- function(x) {
  x <- tolower(trimws(as.character(x)))
  x <- gsub("^https?://", "", x)
  x <- gsub("^www\\.", "", x)
  x <- gsub("/+$", "", x)
  dplyr::na_if(x, "")
}
