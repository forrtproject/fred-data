# Add source variable to track data origin. Both source sheets use
# `out_quote_source`; the rest of the pipeline (and downstream COS/SCORE
# branches) use `outcome_quote_source`, so harmonise here.
tag_source <- function(df, type_label, source_label) {
  df %>%
    rename_with(~ "outcome_quote_source", any_of("out_quote_source")) %>%
    mutate(type = type_label, source = source_label)
}
if (nrow(reproductions) == 0) {

    tag_source(replications, "replication", "replications"),
    tag_source(reproductions, "reproduction", "reproductions")

# Clean scraped DOIs: strip prefixes, whitespace, and trailing garbage phrases
# (e.g. "10.1002/pits.22106digital object identifier (doi)",
#       "10.1016/j.learninstruc.2018.04.010 get rights and content").
# DOIs cannot contain whitespace, so we drop anything from the first space on.
clean_doi <- function(x) {
  x <- tolower(trimws(as.character(x)))
  x <- gsub("^(https?://)?(dx\\.)?doi\\.org/", "", x)
  x <- gsub("^doi:\\s*", "", x)
  x <- sub("\\s*digital\\s*object\\s*identifier.*$", "", x)
  x <- sub("\\s*get\\s*rights\\s*and\\s*content.*$", "", x)
  x <- sub("\\s.*$", "", x)
  x <- trimws(x)
  dplyr::na_if(x, "")
}

flora_data <- flora_raw %>%
  select(all_of(flora_cols)) %>%
  mutate(
    doi_o = clean_doi(doi_o),
    doi_r = clean_doi(doi_r)
  ) %>% # remove redundant DOI urls
  mutate(
    url_r = if_else(!is.na(doi_r) & str_detect(tolower(url_r), if_else(is.na(doi_r), fixed("dummy_val"), fixed(doi_r))), 
                    NA_character_, url_r)
  )

# Normalize URLs before dedup so that e.g. https://osf.io/abc/ and osf.io/abc match
norm_url <- function(x) {
  x <- tolower(trimws(as.character(x)))
  x <- gsub("^https?://", "", x)
  x <- gsub("^www\\.", "", x)
  x <- gsub("/+$", "", x)
  dplyr::na_if(x, "")
}
