# Normalize URLs before dedup so that e.g. https://osf.io/abc/ and osf.io/abc match
norm_url <- function(x) {
  x <- tolower(trimws(as.character(x)))
  x <- gsub("^https?://", "", x)
  x <- gsub("^www\\.", "", x)
  x <- gsub("/+$", "", x)
  dplyr::na_if(x, "")
}
