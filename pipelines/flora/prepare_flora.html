Data Preparation

# Select relevant fields for FLoRA
flora_cols <- c("doi_o", "ref_o", "study_o", "doi_r", "ref_r",  "url_r",
                "abstract_r", "outcome", "outcome_quote", "outcome_quote_source", "type", "source")

flora_data <- flora_raw %>%
  select(all_of(flora_cols)) %>%
  # Normalize DOIs: lowercase and strip common prefixes

  mutate(
    doi_o = tolower(doi_o),
    doi_r = tolower(doi_r),
    # Strip "doi:" prefix and URL prefixes
    doi_o = gsub("^(https?://)?(dx\\.)?doi\\.org/", "", doi_o),
    doi_r = gsub("^(https?://)?(dx\\.)?doi\\.org/", "", doi_r),
    doi_o = gsub("^doi:", "", doi_o),
    doi_r = gsub("^doi:", "", doi_r)
  )

# ---- Step 7 stats helpers ----
has_text <- function(x) !is.na(x) & nzchar(trimws(as.character(x)))
