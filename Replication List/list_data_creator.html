LibRe²

The successor to the meta-analyses-focused FORRT Replication Database (FReD) is a library of replications and reproductions that does not include methodological features such as sample and effect sizes.

Notes: This is heavily based on the fred-mini creator. Currently, it includes code to download crossref keywords. It is, however, also possible to attach the already downloaded keywords instead of getting them all from scratch.

get_doi_data <- function(doi) {
  #100 at a time to avoid URL too long CR error
  doi <- str_replace_all(doi, "https?://doi.org/", "")
  doi <- str_subset(doi, "^10\\.\\d+/\\S+$")
  doi <- unique(doi)
  
  message(glue("Fetching data for {length(doi)} DOIs from CrossRef..."))
  
  if (length(doi) == 0) return(tibble())
  
  for (i in seq(1, length(doi), by = 100)) {
    doisub <- doi[i:min(i+99, length(doi))]
    names(doisub) <- rep("doi", length(doisub))
    res <- cr_works(filter = doisub, limit = 100)
    message(glue("  Retrieved {nrow(res$data)} records (of {length(doisub)} requested)"))
    if(i == 1) out <- res$data else out <- bind_rows(out, res$data)
  }
  message(glue("Done. Retrieved {nrow(out)} records in total."))
  
  return(out)
}

safe_year <- function(df) {
  pub_cols <- c("published.print", "published.online", "published")
  existing <- pub_cols[pub_cols %in% names(df)]

  if (length(existing) == 0) {
    return(NA_integer_)
  }

  year_str <- dplyr::coalesce(!!!df[existing])
  as.integer(stringr::str_sub(as.character(year_str), 1, 4))
}

apa_reference_md <- function(doi, title, author_json, journal, year, volume, issue, pages) {
  n <- length(doi)

  map_chr(seq_len(n), function(i) {
    # --- parse authors safely ---
    authors <- tryCatch(fromJSON(author_json[i]), error = function(e) NULL)

    fmt_author <- function(given, family) {
      initials <- if (!is.na(given) && nzchar(given)) {
        parts <- strsplit(given, "\\s+")[[1]]
        paste0(substr(parts, 1, 1), collapse = ". ")
      } else ""
      paste0(family, ", ", initials, ".")
    }

    if (is.null(authors) || nrow(authors) == 0) {
      author_str <- "(n.a.)"
    } else {
      author_strs <- pmap_chr(authors, \(given, family, sequence, ORCID)
        fmt_author(given, family))
      n_auth <- length(author_strs)
      author_str <-
        if (n_auth == 1) author_strs
        else if (n_auth == 2) paste(author_strs, collapse = " & ")
        else paste0(paste(author_strs[-n_auth], collapse = ", "), ", & ", author_strs[n_auth])
    }

    # --- APA-style markdown (title not bold) ---
    glue(
      "{author_str} ({year[i]}). {title[i]}. *{journal[i]}*, *{volume[i]}*({issue[i]}), {pages[i]}. [https://doi.org/{doi[i]}](https://doi.org/{doi[i]})"
    )
  })
}

bibtex_reference <- function(doi, title, author_json, journal, year, volume, issue, pages) {
  n <- length(doi)

  map_chr(seq_len(n), function(i) {
    # --- parse authors safely ---
    authors <- tryCatch(fromJSON(author_json[i]), error = function(e) NULL)

    fmt_author_bib <- function(given, family) {
      if (!is.na(given) && nzchar(given)) {
        paste0(family, ", ", given)
      } else {
        family
      }
    }

    if (is.null(authors) || nrow(authors) == 0) {
      author_str <- "Unknown"
    } else {
      author_strs <- pmap_chr(authors, \(given, family, sequence, ORCID)
        fmt_author_bib(given, family))
      author_str <- paste(author_strs, collapse = " and ")
    }

    key <- str_remove_all(paste0(str_to_lower(word(author_strs[1], 1)), year[i]), "[^a-z0-9]")

    glue(
"@article{{{key},
  author = {{{author_str}}},
  title = {{{title[i]}}},
  journal = {{{journal[i]}}},
  year = {{{year[i]}}},
  volume = {{{volume[i]}}},
  number = {{{issue[i]}}},
  pages = {{{pages[i]}}},
  doi = {{{doi[i]}}}
}}"
    )
  })
}

get_openalex_keywords <- function(doi, mailto = NULL, fallback_to_concepts = TRUE, n_concepts = 10) {
  base_url <- "https://api.openalex.org/works/"
  # Encode only the DOI part so slashes become %2F but keep "doi:" literal
  encoded_doi <- URLencode(doi, reserved = TRUE)
  url <- paste0(base_url, "doi:", encoded_doi)

  q <- list(select = "id,doi,display_name,keywords,concepts")
  if (!is.null(mailto)) q$mailto <- mailto

  resp <- GET(url, query = q, user_agent("R script (OpenAlex lookup)"))
  if (http_error(resp)) {
    stop(sprintf("HTTP %s: %s", status_code(resp), content(resp, as = "text", encoding = "UTF-8")))
  }
  x <- fromJSON(content(resp, as = "text", encoding = "UTF-8"), simplifyVector = TRUE)

  # Try explicit keywords first
  kws <- NULL
  if (!is.null(x$keywords)) {
    if (is.character(x$keywords)) {
      kws <- x$keywords
    } else if (is.data.frame(x$keywords)) {
      nm <- names(x$keywords)
      pick <- intersect(c("display_name", "keyword", "name", "value"), nm)
      if (length(pick)) kws <- unique(na.omit(as.character(x$keywords[[pick[1]]])))
    } else if (is.list(x$keywords)) {
      # list of lists
      kws <- unique(na.omit(as.character(unlist(lapply(
        x$keywords,
        function(k) k[["display_name"]] %||% k[["keyword"]] %||% k[["name"]] %||% k
      )))))
    }
  }

  # Fallback to top-N concepts if no explicit keywords
  if ((is.null(kws) || length(kws) == 0) && fallback_to_concepts && !is.null(x$concepts)) {
    cons <- x$concepts
    if (is.data.frame(cons) && "display_name" %in% names(cons)) {
      if ("score" %in% names(cons)) {
        cons <- cons[order(cons$score, decreasing = TRUE), , drop = FALSE]
      }
      kws <- head(unique(as.character(cons$display_name)), n_concepts)
    } else if (is.list(cons)) {
      # list of concepts
      dn <- unlist(lapply(cons, function(cn) cn[["display_name"]]))
      sc <- unlist(lapply(cons, function(cn) cn[["score"]]))
      ord <- order(ifelse(is.na(sc), 0, sc), decreasing = TRUE)
      kws <- head(unique(as.character(dn[ord])), n_concepts)
    }
  }

  # Clean up
  kws <- unique(trimws(kws))
  kws[nzchar(kws)]
}
